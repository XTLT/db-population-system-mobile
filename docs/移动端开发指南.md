# Flutter 移动端开发指南

## 📋 项目说明

本项目为人口通的移动端应用，连接现有后端 API。

- **后端位置**：`population-system/server/`（当前仓库 db-population-system）
- **移动端位置**：`population-system/mobile/`（Flutter 项目）

### 1. 配置管理

**文件：`lib/config/app_config.dart`**

```dart
/// 应用配置类
/// 支持动态修改 API 域名
import 'package:shared_preferences/shared_preferences.dart';

class AppConfig {
  // 默认域名（可被覆盖）
  static String apiBaseUrl = 'https://api.yourdomain.com';
  
  // 备用域名列表
  static List<String> backupUrls = [];
  
  // 应用信息
  static const String appName = '人口通';
  static const String appVersion = '1.0.0';
  
  /// 从本地存储加载配置
  static Future<void> loadConfig() async {
    final prefs = await SharedPreferences.getInstance();
    
    // 加载 API 域名
    final savedUrl = prefs.getString('api_base_url');
    if (savedUrl != null && savedUrl.isNotEmpty) {
      apiBaseUrl = savedUrl;
    }
    
    // 加载备用域名
    final savedBackupUrls = prefs.getStringList('backup_urls');
    if (savedBackupUrls != null && savedBackupUrls.isNotEmpty) {
      backupUrls = savedBackupUrls;
    }
  }
  
  /// 更新 API 域名
  static Future<void> updateBaseUrl(String newUrl) async {
    final prefs = await SharedPreferences.getInstance();
    await prefs.setString('api_base_url', newUrl);
    apiBaseUrl = newUrl;
  }
  
  /// 添加备用域名
  static Future<void> addBackupUrl(String url) async {
    if (!backupUrls.contains(url)) {
      backupUrls.add(url);
      final prefs = await SharedPreferences.getInstance();
      await prefs.setStringList('backup_urls', backupUrls);
    }
  }
  
  /// 恢复默认域名
  static Future<void> resetToDefault() async {
    final prefs = await SharedPreferences.getInstance();
    await prefs.remove('api_base_url');
    apiBaseUrl = 'https://api.yourdomain.com';
  }
  
  /// 获取完整的 API URL
  static String getApiUrl(String path) {
    final base = apiBaseUrl.endsWith('/') 
        ? apiBaseUrl.substring(0, apiBaseUrl.length - 1) 
        : apiBaseUrl;
    return '$base$path';
  }
}
```

### 2. HTTP 客户端

**文件：`lib/api/api_client.dart`**

```dart
import 'package:dio/dio.dart';
import '../config/app_config.dart';

class ApiClient {
  static final ApiClient _instance = ApiClient._internal();
  factory ApiClient() => _instance;
  
  late Dio _dio;
  
  ApiClient._internal() {
    _dio = Dio(BaseOptions(
      baseUrl: AppConfig.apiBaseUrl,
      connectTimeout: const Duration(seconds: 10),
      receiveTimeout: const Duration(seconds: 10),
      headers: {
        'Content-Type': 'application/json',
      },
    ));
    
    // 添加拦截器
    _dio.interceptors.add(InterceptorsWrapper(
      onRequest: (options, handler) {
        // 添加 Token
        final token = _getToken();
        if (token != null) {
          options.headers['Authorization'] = 'Bearer $token';
        }
        
        // 使用当前域名
        options.baseUrl = AppConfig.apiBaseUrl;
        
        return handler.next(options);
      },
      onError: (error, handler) async {
        // 域名失效，尝试切换备用域名
        if (error.type == DioExceptionType.connectionError ||
            error.type == DioExceptionType.connectionTimeout) {
          
          final newDomain = await _tryBackupDomain();
          if (newDomain != null) {
            // 重试请求
            error.requestOptions.baseUrl = newDomain;
            try {
              final response = await _dio.fetch(error.requestOptions);
              return handler.resolve(response);
            } catch (e) {
              // 重试失败
            }
          }
        }
        
        return handler.next(error);
      },
    ));
  }
  
  Dio get dio => _dio;
  
  String? _getToken() {
    // TODO: 从本地存储读取 Token
    return null;
  }
  
  Future<String?> _tryBackupDomain() async {
    for (final url in AppConfig.backupUrls) {
      if (await _checkDomain(url)) {
        await AppConfig.updateBaseUrl(url);
        return url;
      }
    }
    return null;
  }
  
  Future<bool> _checkDomain(String url) async {
    try {
      final response = await Dio().get(
        '$url/api/health',
        options: Options(
          receiveTimeout: const Duration(seconds: 5),
        ),
      );
      return response.statusCode == 200;
    } catch (e) {
      return false;
    }
  }
}
```

### 2.1 证书固定（Certificate Pinning）

为增强安全性，移动端支持证书固定，只信任服务端的证书。

**部署步骤**:

```bash
# 1. 启动服务端生成证书
cd server && python app_modular.py

# 2. 导出证书到移动端
cd server && python export_cert.py --mobile

# 3. 证书会自动保存到 mobile/assets/certs/server.crt

# 4. 重新构建 APP
cd mobile && flutter build apk
```

**验证方式**:
- 有证书文件：启用证书固定验证 ✓
- 无证书文件：跳过验证（仅开发环境）⚠️

**文件位置**:
```
mobile/assets/certs/
├── README.md        # 部署说明
└── server.crt       # 服务端证书（需从服务端复制）
```

**注意**: 证书固定需要在构建 APP 前配置，证书文件会打包到 APP 中。更新证书需要重新发布 APP 版本。

更多详情请参考 [安全配置指南](安全配置指南.md)

### 3. 域名服务

**文件：`lib/services/domain_service.dart`**

```dart
import 'package:dio/dio.dart';
import '../config/app_config.dart';

class DomainService {
  /// 检查域名可用性
  static Future<bool> checkDomain(String url) async {
    try {
      final response = await Dio().get(
        '$url/api/health',
        options: Options(
          receiveTimeout: const Duration(seconds: 5),
        ),
      );
      return response.statusCode == 200;
    } catch (e) {
      return false;
    }
  }
  
  /// 测试当前域名
  static Future<Map<String, dynamic>> testCurrentDomain() async {
    final url = AppConfig.apiBaseUrl;
    final startTime = DateTime.now();
    
    try {
      final response = await Dio().get('$url/api/health');
      final endTime = DateTime.now();
      final duration = endTime.difference(startTime).inMilliseconds;
      
      return {
        'url': url,
        'available': response.statusCode == 200,
        'response_time': duration,
      };
    } catch (e) {
      return {
        'url': url,
        'available': false,
        'response_time': 0,
        'error': e.toString(),
      };
    }
  }
  
  /// 获取所有域名状态
  static Future<List<Map<String, dynamic>>> checkAllDomains() async {
    final results = <Map<String, dynamic>>[];
    
    // 检查主域名
    results.add(await testCurrentDomain());
    
    // 检查备用域名
    for (final url in AppConfig.backupUrls) {
      final startTime = DateTime.now();
      try {
        final response = await Dio().get('$url/api/health');
        final endTime = DateTime.now();
        
        results.add({
          'url': url,
          'available': response.statusCode == 200,
          'response_time': endTime.difference(startTime).inMilliseconds,
        });
      } catch (e) {
        results.add({
          'url': url,
          'available': false,
          'response_time': 0,
        });
      }
    }
    
    return results;
  }
}
```

### 4. 设置页面（域名配置）

**文件：`lib/pages/settings/domain_settings.dart`**

```dart
import 'package:flutter/material.dart';
import '../../config/app_config.dart';
import '../../services/domain_service.dart';

class DomainSettingsPage extends StatefulWidget {
  const DomainSettingsPage({super.key});

  @override
  State<DomainSettingsPage> createState() => _DomainSettingsPageState();
}

class _DomainSettingsPageState extends State<DomainSettingsPage> {
  final _urlController = TextEditingController();
  bool _isTesting = false;
  Map<String, dynamic>? _testResult;
  
  @override
  void initState() {
    super.initState();
    _urlController.text = AppConfig.apiBaseUrl;
  }
  
  @override
  void dispose() {
    _urlController.dispose();
    super.dispose();
  }
  
  Future<void> _testConnection() async {
    setState(() {
      _isTesting = true;
      _testResult = null;
    });
    
    final result = await DomainService.testCurrentDomain();
    
    setState(() {
      _isTesting = false;
      _testResult = result;
    });
  }
  
  Future<void> _updateDomain() async {
    final newUrl = _urlController.text.trim();
    if (newUrl.isEmpty) {
      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(content: Text('域名不能为空')),
      );
      return;
    }
    
    await AppConfig.updateBaseUrl(newUrl);
    
    if (mounted) {
      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(content: Text('域名已更新')),
      );
    }
  }
  
  Future<void> _resetToDefault() async {
    await AppConfig.resetToDefault();
    _urlController.text = AppConfig.apiBaseUrl;
    
    if (mounted) {
      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(content: Text('已恢复默认域名')),
      );
    }
  }
  
  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(
        title: const Text('服务器设置'),
      ),
      body: Padding(
        padding: const EdgeInsets.all(16.0),
        child: Column(
          crossAxisAlignment: CrossAxisAlignment.start,
          children: [
            const Text(
              'API 域名',
              style: TextStyle(fontSize: 16, fontWeight: FontWeight.bold),
            ),
            const SizedBox(height: 8),
            TextField(
              controller: _urlController,
              decoration: const InputDecoration(
                border: OutlineInputBorder(),
                hintText: 'https://api.yourdomain.com',
              ),
            ),
            const SizedBox(height: 16),
            Row(
              children: [
                ElevatedButton(
                  onPressed: _isTesting ? null : _testConnection,
                  child: _isTesting
                      ? const SizedBox(
                          width: 16,
                          height: 16,
                          child: CircularProgressIndicator(strokeWidth: 2),
                        )
                      : const Text('测试连接'),
                ),
                const SizedBox(width: 8),
                ElevatedButton(
                  onPressed: _updateDomain,
                  child: const Text('保存'),
                ),
                const SizedBox(width: 8),
                TextButton(
                  onPressed: _resetToDefault,
                  child: const Text('恢复默认'),
                ),
              ],
            ),
            if (_testResult != null) ...[
              const SizedBox(height: 16),
              Container(
                padding: const EdgeInsets.all(12),
                decoration: BoxDecoration(
                  color: _testResult!['available'] == true
                      ? Colors.green.shade50
                      : Colors.red.shade50,
                  borderRadius: BorderRadius.circular(8),
                ),
                child: Column(
                  crossAxisAlignment: CrossAxisAlignment.start,
                  children: [
                    Text(
                      _testResult!['available'] == true ? '✓ 连接成功' : '✗ 连接失败',
                      style: TextStyle(
                        color: _testResult!['available'] == true
                            ? Colors.green
                            : Colors.red,
                        fontWeight: FontWeight.bold,
                      ),
                    ),
                    if (_testResult!['response_time'] != null)
                      Text('响应时间: ${_testResult!['response_time']}ms'),
                  ],
                ),
              ),
            ],
          ],
        ),
      ),
    );
  }
}
```
